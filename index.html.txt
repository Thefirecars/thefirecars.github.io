    <script>
        const tg = window.Telegram.WebApp;
        tg.ready();
        tg.expand();

        const canvas = document.getElementById('coloring-canvas');
        const ctx = canvas.getContext('2d', { willReadFrequently: true });
        
        let currentPage = 1;
        const totalPages = 4;
        
        // This configuration maps explicitly to your exact mobile file layout names
        const pageImages = {
            1: 'page1.png.png',
            2: 'page2.png.png',
            3: 'page3.png.png',
            4: 'page4.png.png'
        };
let selectedColor =;
        let baseImage = new Image();

        function loadBookPage(num) {
            document.getElementById('page-title').innerText = `Page ${num} of ${totalPages}`;
            baseImage.src = pageImages[num];
            baseImage.onload = () => {
                canvas.width = baseImage.width;
                canvas.height = baseImage.height;
                ctx.drawImage(baseImage, 0, 0);
            };
        }

        document.querySelectorAll('.color-btn').forEach(btn => {
            btn.addEventListener('click', (e) => {
                document.querySelector('.color-btn.active').classList.remove('active');
                e.target.classList.add('active');
                const rgbString = e.target.getAttribute('data-color');
                selectedColor = rgbString.match(/\d+/g).map(Number);
            });
        });

        function changePage(direction) {
            currentPage += direction;
            if (currentPage < 1) currentPage = totalPages;
            if (currentPage > totalPages) currentPage = 1;
            loadBookPage(currentPage);
        }

        canvas.addEventListener('click', (e) => {
            const rect = canvas.getBoundingClientRect();
            const startX = Math.floor((e.clientX - rect.left) * (canvas.width / rect.width));
            const startY = Math.floor((e.clientY - rect.top) * (canvas.height / rect.height));
            floodFill(startX, startY, selectedColor);
        });
function floodFill(startX, startY, fillColor) {
            const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);
            const data = imgData.data;
            const width = imgData.width;
            const height = imgData.height;

            const targetIdx = (startY * width + startX) * 4;
            const startR = data[targetIdx];
            const startG = data[targetIdx+1];
            const startB = data[targetIdx+2];

            if ((startR < 120 && startG < 120 && startB < 120) || 
                (startR === fillColor[0] && startG === fillColor[1] && startB === fillColor[2])) {
                return;
            }

            const queue = [[startX, startY]];

            while (queue.length > 0) {
                const [x, y] = queue.shift();
                const idx = (y * width + x) * 4;

                if (Math.abs(data[idx] - startR) < 30 && Math.abs(data[idx+1] - startG) < 30 && Math.abs(data[idx+2] - startB) < 30) {
                    data[idx] = fillColor[0];
                    data[idx+1] = fillColor[1];
                    data[idx+2] = fillColor[2];
                    data[idx+3] = fillColor[3];

                    if (x > 0) queue.push([x - 1, y]);
                    if (x < width - 1) queue.push([x + 1, y]);
                    if (y > 0) queue.push([x, y - 1]);
                    if (y < height - 1) queue.push([x, y + 1]);
                }
            }
            ctx.putImageData(imgData, 0, 0);
        }

        document.getElementById('buy-premium-btn').addEventListener('click', () => {
            tg.openLink('https://gumroad.com');
        });

        loadBookPage(currentPage);
    </script>
